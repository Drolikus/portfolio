const cart = {
    items: [],
    add(product) {
        const existing = this.items.find(i => i.id === product.id);
        if (existing) {
            existing.qty++;
        } else {
            this.items.push({ ...product, qty: 1 });
        }
        this.updateUI();
    },
    total() {
        return this.items.reduce((sum, i) => 
            sum + i.price * i.qty, 0
        );
    }
};
